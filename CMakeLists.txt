cmake_minimum_required(VERSION 3.24)
project(WhatInput VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
add_library(sqlite STATIC third_party/sqlite3.c)
target_include_directories(sqlite PUBLIC third_party)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
add_library(core STATIC src/core.cpp)
target_include_directories(core PUBLIC src)
target_link_libraries(core PUBLIC sqlite)
if(WIN32)
  add_executable(WhatInput WIN32 src/collector.cpp)
  target_compile_definitions(WhatInput PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(WhatInput PRIVATE core user32 shell32 advapi32 xinput9_1_0 powrprof)
  install(TARGETS WhatInput RUNTIME DESTINATION .)
endif()
option(BUILD_STATS "Build Qt statistics window" ON)
if(BUILD_STATS)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Concurrent Network)
  set(CMAKE_AUTOMOC ON)
  qt_add_executable(WhatInputStats WIN32 src/stats.cpp)
  qt_add_resources(WhatInputStats web PREFIX / FILES web/index.html web/style.css web/app.js)
  target_link_libraries(WhatInputStats PRIVATE core Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Concurrent Qt6::Network)
  target_compile_definitions(WhatInputStats PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  install(TARGETS WhatInputStats RUNTIME DESTINATION .)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE core)
add_test(NAME core_tests COMMAND core_tests)
